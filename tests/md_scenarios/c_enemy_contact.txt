# (c) Hall : Simon reste immobile, un ennemi de type 1 le touche, l'energie baisse de 2,
# l'invulnerabilite et le recul demarrent (research/player-health.md, « Recul au coup recu »).
include boot_to_courtyard.inc
until front_mode == MODES['INTERIOR'] max 3000 RIGHT
until enemies.alive > 0 max 600
expect enemies.energy == 32 # energie pleine avant contact
until enemies.energy < 32 max 600                 # aucune entree : l'ennemi avance jusqu'a Simon
dump contact
shot c-contact
expect enemies.energy == 30 # -2 par contact (ENEMY_CONTACT_DAMAGE)
expect enemies.invuln > 0 # invulnerabilite lancee
expect enemies.recoil > 0 and game.recoil == 1 # recul en cours (C420 = 5, phase 0)
set x_contact game.player.x
set y_contact game.player.y
set dir game.recoil_dir
# Arc de 707D : 16 px de montee, 2 px par image ; les touches sont ignorees.
until game.player.y == y_contact - 16 max 12 LEFT+B
# Touche en x 16 : la salle 0 a une salle a gauche, le port borne alors a x 8 (PROVISOIRE).
expect game.player.x == max(8, min(248, x_contact + 2 * 9 * dir)) # 9 images de montee
shot c-recul
until game.recoil == 3 max 20 LEFT+B
expect game.player.x == max(8, min(248, x_contact + 44 * dir)) and game.player.y == y_contact # 22 pas, 44 px (trace : 136 -> 92)
expect whip.timer == 0 # B ignore pendant le recul
shot c-pose-au-sol
wait 4
dump apres-recul
expect enemies.recoil == 0 and game.recoil == 0 # 4 images au sol puis fin (702A)
expect enemies.energy == 30 # pas de second degat pendant l'invulnerabilite
