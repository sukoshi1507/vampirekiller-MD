# (i) Torche A de la cour, salle 0 : le fouet l'eteint (regle 8450), un petit coeur
# tombe en se balancant (8B22) et Simon le ramasse (847A) : coeurs 0 -> 1
# (trace openMSX scratchpad/drops, plan C : torche touchee a x 44, y 176).
# vk_candles (candles.h) : lit en +4, coeurs en +118 (6 + 8 objets de 14 octets).
include boot_to_courtyard.inc
poke vk_game+0 44
wait 4
expect ram8('vk_candles+4') == 3 # torches A et B allumees
hold B 2
until (ram8('vk_candles+4') & 1) == 0 max 20
expect ram8('vk_candles+4') == 2 # torche A eteinte, B allumee
wait 30
shot i-coeur-tombe
until ram8('vk_candles+6') == 3 max 400 # le coeur est pose (VK_DROP_REST)
shot i-coeur-pose
set hx ram8('vk_candles+7')
expect 64 <= hx <= 112 # pres de la torche
until ram8('vk_candles+118') == 1 max 200 RIGHT
expect ram8('vk_candles+6') == 0 # coeur ramasse
wait 4
shot i-coeur-ramasse
