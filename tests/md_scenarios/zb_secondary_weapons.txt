# (zb) Armes lancees (C416 >= 2, 7116-75BF ; research/secondary-weapons.md), hall salle 0.
# L'objet est pose sous Simon dans le premier emplacement C500 de vk_candles (voir
# t_item_effects) ; C416 = vk_candles+120, coeurs C417 = +118, C701 = +172.
# vk_weapons : emplacement C450 +0..+8 (etat, type, dy, dx, Y, X, motif, compteur, sens),
# C460 +9..+17. C434 (enemy_state+143) protege Simon des ennemis pendant le scenario.
include boot_to_courtyard.inc
until front_mode == MODES['INTERIOR'] max 3000 RIGHT
wait 40                                            # fin de l'etat 9 de la salle 0
poke vk_game+0 136
wait 2
poke vk_candles+118 12                             # 12 coeurs
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x1B
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 2
expect ram8('vk_candles+6') == 0 and ram8('vk_candles+120') == 2 and whip.level == 2 # dague 1B : C416 = 2 (8D70)
expect ram8('vk_candles+178') != 0 # icone de l'arme dans le HUD (8E9A)
poke enemy_state+143 0x96
until whip.timer == 0 max 30
# Dague : B reserve C450 (71BE), lancement a la 4e passe (7286), vitesse 5, Y = C425 - 16.
tap B 1
until ram8('vk_weapons+1') == 2 max 20
expect ram8('vk_weapons+4') == (game.player.y - 16) & 0xFF and ram8('vk_weapons+0') == 2 # 7436
expect ram8('vk_weapons+3') in (5, 251) # dx +-5
set x0 ram8('vk_weapons+5')
wait 4
expect ram8('vk_weapons+5') in ((x0 + 10) & 0xFF, (x0 - 10) & 0xFF) # 2 passes en 4 images
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+10') == 2 max 20
expect ram8('vk_weapons+1') == 2 # deux dagues en vol (C450, C460)
shot daggers
until whip.timer == 0 max 30
hold B 1
expect whip.timer == 0 # aucun emplacement libre : appui ignore (71BE)
until ram8('vk_weapons+1') == 0 and ram8('vk_weapons+10') == 0 max 120
expect ram8('vk_weapons+4') == 0xE0 and ram8('vk_candles+120') == 2 # emplacements vides, la dague reste
# Hache 1C : un seul emplacement, boomerang, rattrapee par Simon (8240) : C416 reste 3.
poke vk_candles+9 0x1C
poke vk_candles+6 3
wait 2
expect ram8('vk_candles+120') == 3 and whip.level == 3 # hache
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+1') == 3 max 20
until ram8('vk_weapons+0') == 3 max 60
expect ram8('vk_weapons+1') == 3 # etat 3 : ralentit puis revient
shot axe
until ram8('vk_weapons+1') == 0 max 120
expect ram8('vk_candles+120') == 3 # rattrapee
# Croix 1D.
poke vk_candles+9 0x1D
poke vk_candles+6 3
wait 2
expect ram8('vk_candles+120') == 4 and whip.level == 4 # croix
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+1') == 4 max 20
wait 20
shot cross
until ram8('vk_weapons+1') == 0 max 200
expect ram8('vk_candles+120') == 4 # rattrapee
# Eau benite 1E (C701 bit 3) : en saut, DROITE nouvellement pressee, 5 coeurs (714D).
poke vk_candles+9 0x1E
poke vk_candles+6 3
wait 2
expect ram8('vk_candles+172') & 0x08 # 8D8D
set h0 ram8('vk_candles+118')
hold RIGHT 1
expect ram8('vk_weapons+10') == 0 and ram8('vk_candles+118') == h0 # pas au sol
tap C 1
expect game.player.mode == 1 # en l'air
hold RIGHT 1
wait 1
expect ram8('vk_weapons+10') == 5 and ram8('vk_candles+118') == h0 - 5 # C461 = 5, -5 coeurs
wait 12
shot holy_water
until ram8('vk_weapons+9') == 3 max 120
wait 4
shot flames
expect ram8('vk_weapons+15') in (0xF4, 0xF8) or ram8('vk_weapons+15') == 0x38 # flammes au sol
until ram8('vk_weapons+10') == 0 max 80
expect ram8('vk_candles+120') == 4 # la croix reste
