until enemies.energy < 32 max 600                 # l'ennemi arrive de la droite

expect game.player.x == 0xF6 and game.recoil == 2 # salle 3, X F6, toujours en l'air (C420 = 5)
expect frame - f_contact <= 12 # pose, puis 5 pas de 2 px (16 -> 6), une passe toutes les 2 images
until game.recoil == 3 max 200                    # apres le gel de l'etat 9, atterrissage en salle 3
expect game.room == 3 and game.player.y == 192 and game.player.x == 0xF6 - 2 * 17
