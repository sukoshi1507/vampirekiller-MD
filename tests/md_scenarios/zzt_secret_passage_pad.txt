# (zzt) Stage 12 secret passage with the keys a player has on the pad (tester: "the teleporter does not work").
# MSX (6DA9, openMSX stage 12 room 0, Simon at 92,176): crouched (C420 = 2) in the zone of the BBCD point and the
# jump key newly pressed (C006 bit 5 = UP on the keyboard, button 2 on a joystick, 4BBB) -> C420 = 7, C42D = 40h,
# room 3 after 64 passes, X and Y kept. DOWN released one pass before UP: Simon stands up and jumps, no passage.
# On the pad, DOWN held + UP cannot be done: the jump button C opens the passage.
include zzt_enter12.inc
until game.stage == 12 and game.room == 0 max 300
poke Enemies_god 1
until game.player.mode == 0 max 300
wait 30
expect ram8('vk_secret') == 1 and ram8('vk_secret+2') == 88 and ram8('vk_secret+3') == 3 # BBCD: room 0 -> 3, X 88
# Walk into the dark arch: the zone is X 88..103 at the floor of the room (Y 176).
until game.player.x >= 94 max 300 RIGHT
wait 4
expect 88 <= game.player.x < 104 and game.player.y == 176
# What a pad can do with the D-pad alone: DOWN, back to neutral, UP. No passage (MSX: plan sec_rel).
hold DOWN 12
wait 3
hold UP 6
wait 20
expect ram8('vk_secret+4') == 0 and game.room == 0 # still in room 0
# DOWN held, then C (jump): the passage opens.
hold DOWN 12
expect game.player.crouched == 1
hold DOWN+C 4
expect ram8('vk_secret+4') == 7 # C420 = 7
set x0 game.player.x
shot zzt-passage
until game.room == 3 max 200
expect game.stage == 12 and game.player.x == x0 and game.player.y == 176 # B98C: only D001 changes
until front_mode == MODES['INTERIOR'] and ram8('vk_secret') == 1 max 200
wait 60
expect game.room == 3 and ram8('vk_secret+3') == 0 # BBCD: room 3 -> 0 at the same point
# And back, the button first held a few frames like a hand does.
hold DOWN 20
hold DOWN+C 8
expect ram8('vk_secret+4') == 7
until game.room == 0 max 200
expect game.stage == 12 and game.player.x == x0
wait 30
shot zzt-retour
