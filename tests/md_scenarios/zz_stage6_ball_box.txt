# scratchpad/stage6boss/out/A (Simon immobile en (246, 160)) : objet en (90h, 90h) a l'image 1230, premiere boule a la passe 60.
include zy_enter6.inc
wait 30
poke items+0 1
poke items+1 1

poke vk_game+0 34
poke vk_game+1 128
poke vk_game+2 1

until game.room == 5 max 400 LEFT
expect game.player.x == 246 and game.player.y == 128 # trace : (246, 128), puis chute a 160
until game.player.y == 160 max 100
expect ram8('vk_boss') == 1 and ram8('vk_boss+56') == 1 and ram8('vk_boss+3') == 0
expect pixel(140, 125) != (0, 0, 0) # le buste (128, 112) est encore la
set t0 frame
until ram8('vk_boss+3') == 1 max 400
set t1 frame
expect 220 <= t1 - t0 <= 250 # CE02 = 78h passes (trace : 242 images apres CE01 = 1, plus le rechargement de la salle)
expect ram16('vk_boss+8') == 0x9030 and ram16('vk_boss+10') == 0x90F8 # trace 1230 : Y 9030h, X 90F8h

expect ram16('vk_boss+8') == 0x9080 and ram16('vk_boss+10') == 0x91F0 # trace 1232
wait 2
expect ram16('vk_boss+8') == 0x90F0 and ram16('vk_boss+10') == 0x92E8 # trace 1234

# Premiere boule : 60 passes apres la creation, depuis (X, Y - 16) de l'objet, visee A0E5 (vy 200, vx 1000 contre Simon (246, 160)).
until ram8('vk_boss+36') == 1 max 400

expect ram16('vk_boss+38') == 0x82C8 and ram16('vk_boss+40') == 0xCCE8 # trace 1348 : Y 82C8h, X CCE8h
expect ram16('vk_boss+42') == 200 and ram16('vk_boss+44') == 1000
expect pixel(140, 125) == (0, 0, 0) # buste efface (HMMV), loin de l'objet

# Contact de la boule : energie 32 -> 31 (trace : image 1348 -> 1364).
until enemies.energy == 31 max 60
expect 12 <= frame - t2 <= 20

poke vk_boss+43 0
poke vk_boss+44 0
