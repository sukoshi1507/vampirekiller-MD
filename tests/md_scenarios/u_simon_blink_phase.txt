# (u) Phase du clignotement de Simon apres un ramassage de 09 (C434) ou 08 (C43A), hall salle 0
# (trace openMSX tools/run_simon_blink_trace.py, research/castle-candles.md 9.8).
# MSX : 790C remplit D400 apres la baisse des minuteries ; l'objet est pris plus loin dans la
# passe (7FE2), la silhouette commence a la passe suivante si C003 y est impair : ramassage
# dans une passe paire -> 75 passes (150 images), impaire -> 74 passes (148 images).
# Port : enemies.tick = C003 (+1 a l'image de passe, itemPhase 0), couleur figee a la passe ;
# pal() = palette du sprite 0 calculee a l'image (PAL3 = silhouette), affichee a l'image
# suivante (red() : pixels rouges 020E de la capture). Image 0 = image du ramassage.
include boot_to_courtyard.inc
until front_mode == MODES['INTERIOR'] max 3000 RIGHT
wait 40
poke vk_game+0 136
wait 2
set pal lambda: (ram16('vdpSpriteCache+4') >> 13) & 3
set red lambda: len([c for c in colors(game.player.x - 8, game.player.y - 31, game.player.x + 16, game.player.y + 1) if c == (238, 0, 32)]) > 40
# 09 dans une passe paire : l'image suivante est une image de passe (itemPhase 1 -> 0) ou
# tick devient pair.
until enemies.itemPhase == 1 and enemies.tick % 2 == 1 max 10
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x09
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 1
expect ram8('vk_candles+171') == 0x09 and enemies.itemPhase == 0 and enemies.tick % 2 == 0 # 09 pris a une passe paire
set acc []
until acc.append((pal(), red())) or len(acc) == 306 max 310
set on [i for i, a in enumerate(acc) if a[0] == 3]
expect on[0] == 2 and on[-1] == 299 and len(on) == 150 # passes 1, 3 ... 149 apres le ramassage
expect [i for i, a in enumerate(acc) if a[1]] == [i + 1 for i in on] # silhouette rouge a l'image suivante
# 08 dans une passe impaire : rien a la passe suivante (paire), 74 passes.
until enemies.killTimer == 0 max 20
until enemies.itemPhase == 1 and enemies.tick % 2 == 0 max 10
poke vk_candles+9 0x08
poke vk_candles+6 3
wait 1
expect ram8('vk_candles+171') == 0x08 and enemies.itemPhase == 0 and enemies.tick % 2 == 1 # 08 pris a une passe impaire
set acc []
until acc.append(pal()) or len(acc) == 306 max 310
set on [i for i, a in enumerate(acc) if a == 3]
expect on[0] == 4 and on[-1] == 297 and len(on) == 148 # passes 2, 4 ... 148 apres le ramassage
expect ram8('simon_flash_color') == 0x0E # 7924 : couleur 0Eh
