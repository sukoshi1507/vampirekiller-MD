# (zzzf) Mode graphique 16 bits (research/gfx16.md, section "Objets 16 bits") : bougies, objets, armes lancees, flammes et butin
# en 16 bits, dessines avec PAL1 (palette du HUD : entrees 1, 2, 8, 14, 15 inchangees, les autres sont les couleurs des objets).
# Salle 5 du hall : bougie (64,144) et bougie (144,128) ; dague 1B posee sous Simon (comme zb), flamme de mort du zombie (comme zzh).
include boot_to_courtyard.inc
poke vk_gfx16 1
until front_mode == MODES['INTERIOR'] max 3000 RIGHT
poke vk_debug_keys 1                               # debug shortcuts are off by default
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
hold A 3
hold A+RIGHT 3
wait 10
expect game.room == 5 # salle 5 du hall
wait 30
expect ram8('vk_gfx16_on') == 1
shot zzzf-bougies
# Le HUD lit PAL1 : ses entrees 1 (peau), 8 (rouge) et 14 (blanc) gardent leurs couleurs (barres, texte).
set hud colors(8, 8, 122, 28)
expect (238, 170, 139) in hud and (238, 238, 238) in hud and (238, 0, 32) in hud # barre du joueur, texte, barre de l'ennemi
set pal1 lambda: [(ram16('vdpSpriteCache+%d' % (8 * i + 4)) >> 13) & 3 for i in range(80)]
# Les deux bougies murales : chandeliers sombres 16 bits, sprites en PAL1.
set cc sorted(colors(64, 144, 80, 160))
expect (65, 32, 0) in cc and len(cc) >= 6 # brun du chandelier
expect 1 in pal1() # sprites de PAL1
# Bougie (64,144), objet 00 : saut avec fouet, flamme F4/F8 (flamme de mort 16 bits, PAL1).
poke vk_game+0 50
wait 4
tap C 1
wait 6
hold B 2
until (ram8('vk_candles+4') & 1) == 0 max 40
wait 6
shot zzzf-flamme-bougie
set fc sorted(colors(56, 136, 88, 168))
expect (238, 170, 0) in fc and (139, 0, 0) in fc # flamme 16 bits (or, rouge sombre)
until ram8('vk_candles+6') == 0 max 40
# Bougie (144,128), objet 1A : boule de flamme (butin des bougies, 12 x 12), puis le fouet a chaine tombe au sol.
until game.player.mode == 0 max 80
poke vk_game+0 124
wait 4
tap C 1
wait 14
hold B 2
until (ram8('vk_candles+4') & 2) == 0 max 40
wait 8
shot zzzf-boule
expect 1 in pal1() # boule 12 x 12 en PAL1
until ram8('vk_candles+6') == 3 max 300
shot zzzf-fouet-sol
until ram8('vk_candles+120') == 1 max 200 RIGHT
expect ram8('vk_candles+6') == 0 # ramasse
# Objet pose au sol (bottes 0C, comme zb : emplacement C500 de vk_candles), vu avant d'etre ramasse.
poke vk_candles+7 100
poke vk_candles+8 176
poke vk_candles+9 0x0C
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 4
shot zzzf-objet-sol
# Ramasse : l'objet entre dans la case des objets du HUD (sprite 16 bits au-dessus du cadre MSX).
poke vk_game+0 108
wait 3
expect ram8('vk_candles+6') == 0
shot zzzf-objet-hud
set bt colors(176, 8, 250, 28)
expect (238, 238, 238) in bt and (205, 206, 238) in bt # bottes 16 bits dans la case du HUD
# Dague 1B : icone de l'arme dans le HUD, puis deux dagues lancees.
poke vk_candles+118 12                             # 12 coeurs
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x1B
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
poke vk_game+0 136
wait 4
expect ram8('vk_candles+120') == 2 and ram8('vk_candles+328') != 0 # dague, icone de l'arme
shot zzzf-arme-hud
set wi colors(128, 8, 150, 30)
expect (0, 68, 205) in wi and (65, 101, 139) in wi and (238, 0, 32) in wi # icone de la dague 16 bits, cadre MSX rouge de la case
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+1') == 2 max 20
wait 3
shot zzzf-dague-lancee
# Le zombie touche par la dague : flamme de mort F4/F8 16 bits.
until enemies.enemy[0].flame == 1 max 120
wait 2
shot zzzf-flamme-mort
set fm colors(172, 164, 190, 180)
expect (238, 170, 0) in fm and (139, 0, 0) in fm # flamme de mort 16 bits
expect 1 in pal1()
# Hache 1C, croix 1D (icone en diagonale) et eau benite 1E : image de l'arme en vol (PAL1), puis la flamme de l'eau benite.
until ram8('vk_weapons+1') == 0 and ram8('vk_weapons+10') == 0 max 200
until whip.timer == 0 max 60
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x1C
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 2
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+1') == 3 max 20
wait 6
shot zzzf-hache
expect 1 in pal1() # hache en PAL1
until ram8('vk_weapons+0') == 0 max 200
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x1D
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 2
shot zzzf-croix-icone
until whip.timer == 0 max 30
tap B 1
until ram8('vk_weapons+1') == 4 max 20
wait 6
shot zzzf-croix
poke vk_candles+118 12
until ram8('vk_weapons+0') == 0 max 300
poke vk_candles+7 128
poke vk_candles+8 176
poke vk_candles+9 0x1E
poke vk_candles+10 0xFF
poke vk_candles+11 0
poke vk_candles+6 3
wait 2
tap C 1
expect game.player.mode == 1 # en l'air
hold RIGHT 1
wait 1
expect ram8('vk_weapons+10') == 5 # fiole lancee
wait 12
shot zzzf-eau-benite
until ram8('vk_weapons+9') == 3 max 120
wait 4
shot zzzf-eau-benite-flamme
