# (zzq) Stage 10 complet a la manette SANS mode invulnerable (Enemies_god coupe par zzm_enter10.inc : un `tap A` de plus apres le Konami), de l'entree a la porte, sans aucune pose
# de position ni de vie. Variante de zzq_stage10_route.txt (qui reste joue en invulnerable) ou les attentes chronometrees sont remplacees par des attentes conditionnelles
# qui fouettent ce qui s'approche (zzq_guard.inc, zzq_walk.inc, zzq_walkl.inc, zzq_clear.inc) : Simon arrive a la porte vivant avec de l'energie en reserve.
# Reproductible : memes entrees depuis le reset = meme partie (les ennemis dependent seulement de la minuterie de la salle et de la position de Simon, pas d'un hasard non rejoue).
# Salle 0 : l'ennemi volant (type 2, une passe toutes les ~160 images, de droite a gauche puis de gauche a droite) ne se fouette bien que par l'avant et seulement au milieu de
#   sa vague (y <= ~122) : on l'attend au bord du sol (X 98) et on le fouette a 31 px ; celui qui vient de derriere est fouette depuis la plateforme (saut sur elle quand X <= 166).
# Salles 1 a 5 et 8 : hommes-poissons (type 17, 2 coups de fouet) qui sautent de l'eau, marchent sur les rebords (y 80) et tombent aux memes endroits (X 85/108/125/157) ; on les
#   attend ou ils ne tombent pas, on les fouette quand ils viennent, on ne saute sur un bloc que quand la salle est vide d'hommes-poissons. Boules de feu (type 10, 1 point) : fouettees.
include zzm_enter10.inc
until game.player.mode == 0 max 300
wait 10
expect ram8('Enemies_god') == 0 and enemies.energy == 32 # pas d'invulnerabilite, energie pleine
set lives0 health.lives
set ct (17,)
set ty 36 # tolerance verticale (pixels) entre Simon et la cible pour fouetter
set gt (2, 17, 10) # types a fouetter : ennemi volant, homme-poisson, boule de feu
until game.player.x >= 98 max 200 RIGHT
until enemies.enemy[0].active == 1 max 200
repeat 80 zzq_guard.inc
until ram8('vk_platforms+2') <= 166 max 300
hold C+RIGHT 1
hold RIGHT 47
repeat 600 zzq_guard_platform.inc
until game.player.x >= 205 max 100 RIGHT
hold C+RIGHT 1
hold RIGHT 60
expect game.stage == 10 and game.room == 1 and enemies.energy == 32 # salle 0 passee sans degat

hold C+RIGHT 1
hold RIGHT 60
set tx 90
repeat 300 zzq_walk.inc

hold C+RIGHT 1
until game.player.y == 112 and game.player.mode == 0 max 80 RIGHT
until any(e.active and e.type == 17 for e in enemies.enemy) max 200
repeat 400 zzq_clear.inc
set tx 152
repeat 100 zzq_walk.inc
hold C+RIGHT 1
hold RIGHT 60
set tx 226

hold RIGHT 45

hold RIGHT 67

hold RIGHT 46

repeat 606 zzq_guard.inc
hold RIGHT 40
until game.player.x >= ram8('vk_platforms+2') + 18 max 100 RIGHT
until game.player.x >= 86 max 400
hold B 3
until game.player.x >= 100 max 100
until ram8('vk_platforms+2') >= 185 max 800
hold RIGHT 80
expect game.stage == 10 and game.room == 4 and game.player.x <= 40 and game.player.y == 144 and enemies.energy == 32 # bloc de la salle 3 casse, aucun degat
set troom 4
set tx 122
repeat 300 zzq_walk.inc
set pmax 150
repeat 300 zzq_guard_pmax.inc
hold C+RIGHT 1
hold RIGHT 70
hold DOWN 40
until ram8('vk_platforms+2') >= 189 max 800 DOWN

hold RIGHT 50

set tx 120
repeat 100 zzq_walk.inc
hold C+RIGHT 1
until game.player.y == 112 and game.player.mode == 0 max 80 RIGHT
until any(e.active and e.type == 17 for e in enemies.enemy) max 200
repeat 400 zzq_clear.inc
set tx 152
repeat 100 zzq_walk.inc
hold C+RIGHT 1
hold RIGHT 60
set tx 255
