# (zzzc) Mode 16 bits : titre, carte du stage (objet 0F) et fin en 16 bits (research/gfx16.md).
until front_mode == MODES['TITLE'] max 2000
shot zzzc-titre-msx
set msx_band pixel(100, 60)
set msx_strip pixel(8, 12)
tap DOWN
expect ram8('vk_gfx16') == 1
wait 4
shot zzzc-titre-16
expect pixel(4, 12) != msx_strip # film : une case sur deux, perforation
expect pixel(100, 60) != msx_band # bande bleue plus sombre
expect pixel(8, 200) == (0, 0, 0) # le bas reste noir
tap DOWN
wait 4
expect ram8('vk_gfx16') == 0
expect pixel(4, 12) == msx_strip and pixel(100, 60) == msx_band # retour MSX identique
tap DOWN
wait 4
expect ram8('vk_gfx16') == 1
tap START
tap START
until front_mode == MODES['COURTYARD'] max 500
wait 60
poke vk_candles+322 0x80
poke vk_candles+329 3
set room0 pixel(30, 60)
tap A
wait 40
shot zzzc-carte-16
expect pixel(30, 60) != room0
expect pixel(30, 60) == pixel(225, 60) and pixel(30, 60) != pixel(10, 60) # panneau x 24..231
expect pixel(22, 100) != (0, 0, 0) and pixel(8, 100) == (0, 0, 0) # cadre dore autour du panneau
expect pixel(150, 22) != pixel(30, 60) # la case du HUD garde son fond
tap A
wait 40
expect pixel(30, 60) == room0 # la salle est revenue
shot zzzc-carte-retour
# Fin : le tableau de Castlevania-MD assombri sur le plan B derriere le defilement du texte (plan A).
